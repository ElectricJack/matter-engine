# Fixture: graphics. Allowed to depend on contracts only. FX_GRAPHICS_SDK adds
# a backend that needs the (never installed) FxSdk, standing in for Vulkan.
cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(fx_graphics LANGUAGES CXX)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/cmake/FxGraphicsOptions.cmake")
if(NOT TARGET fx_contracts)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../FxContracts" "${CMAKE_BINARY_DIR}/fx_contracts")
endif()
add_library(fx_graphics STATIC src/fx_draw.cpp)
target_include_directories(fx_graphics PUBLIC include)
target_link_libraries(fx_graphics PUBLIC fx_contracts)
if(FX_GRAPHICS_SDK)
  list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/cmake")
  find_package(FxSdk REQUIRED)
  add_library(fx_graphics_sdk STATIC src/fx_sdk_backend.cpp)
  target_include_directories(fx_graphics_sdk PUBLIC "${FxSdk_INCLUDE_DIR}")
  target_link_libraries(fx_graphics_sdk PUBLIC fx_graphics)
endif()
