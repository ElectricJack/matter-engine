cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(MatterComputeDevice LANGUAGES C CXX)
endif()
if(NOT TARGET matter_contracts)
    set(_compute_testing ${BUILD_TESTING})
    set(BUILD_TESTING OFF)
    add_subdirectory(../MatterContracts "${CMAKE_CURRENT_BINARY_DIR}/contracts")
    set(BUILD_TESTING ${_compute_testing})
endif()
add_library(matter_compute_device STATIC src/device.cpp)
target_include_directories(matter_compute_device PUBLIC include)
target_link_libraries(matter_compute_device PUBLIC matter_contracts)
target_compile_features(matter_compute_device PUBLIC cxx_std_17)
option(MATTER_COMPUTE_VULKAN "Build the neutral Vulkan compute device" OFF)
if(MATTER_COMPUTE_VULKAN)
    find_package(Vulkan REQUIRED)
    add_library(matter_compute_vulkan STATIC src/vulkan.cpp)
    target_link_libraries(matter_compute_vulkan PUBLIC matter_compute_device Vulkan::Vulkan)
    if(MSVC)
        target_compile_options(matter_compute_vulkan PRIVATE /W4 /WX /permissive- /utf-8)
    else()
        target_compile_options(matter_compute_vulkan PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endif()
