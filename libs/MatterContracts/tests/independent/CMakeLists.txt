cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(MatterContractGate LANGUAGES C CXX)
    # This consumer requires no extraction target, runtime, or monolith.
    set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
    add_subdirectory(../.. contracts)
    enable_testing()
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME gate_manifest COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/../gate_manifest.py)
add_library(gate_callee STATIC producer.cpp)
target_link_libraries(gate_callee PUBLIC matter_contracts)
add_executable(gate_native native.cpp)
target_link_libraries(gate_native PRIVATE gate_callee)
add_executable(gate_graphics graphics.cpp)
target_link_libraries(gate_graphics PRIVATE matter_contracts)
add_executable(gate_c_abi mixed_abi.cpp ../abi_producer.c)
target_include_directories(gate_c_abi PRIVATE ../../include)
target_compile_features(gate_c_abi PRIVATE cxx_std_17)
set_target_properties(gate_c_abi PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
foreach(fixture gate_native gate_graphics gate_c_abi)
    add_test(NAME ${fixture} COMMAND ${fixture})
endforeach()
