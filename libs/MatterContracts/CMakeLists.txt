cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(MatterContracts VERSION 1.0.0 LANGUAGES C CXX)
    include(CTest)
endif()
add_library(matter_contracts STATIC src/value.cpp src/codec.cpp src/validation.cpp src/digest.cpp src/services.cpp)
target_compile_features(matter_contracts PUBLIC cxx_std_17)
target_include_directories(matter_contracts PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/include ${CMAKE_CURRENT_SOURCE_DIR}/../MathLib/include)
if(MSVC)
    target_compile_options(matter_contracts PRIVATE /W4 /WX /permissive- /utf-8)
else()
    target_compile_options(matter_contracts PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
include(cmake/ComponentScaffolds.cmake)
foreach(component GEOMETRY BUILD GRAPHICS TERRAIN ASSET_PIPELINE GEOMETRY_C HOST_QUICKJS)
    option(MATTER_SCAFFOLD_${component} "Register interface-only ${component} scaffold (not extraction acceptance)" OFF)
endforeach()
if(MATTER_SCAFFOLD_GEOMETRY)
    matter_register_component(matter_geometry)
endif()
if(MATTER_SCAFFOLD_BUILD)
    matter_register_component(matter_build)
endif()
if(MATTER_SCAFFOLD_GRAPHICS)
    matter_register_component(matter_graphics)
endif()
if(MATTER_SCAFFOLD_TERRAIN)
    matter_register_component(matter_terrain)
endif()
if(MATTER_SCAFFOLD_ASSET_PIPELINE)
    matter_register_component(matter_asset_pipeline)
endif()
if(MATTER_SCAFFOLD_GEOMETRY_C)
    matter_register_component(matter_geometry_c)
endif()
if(MATTER_SCAFFOLD_HOST_QUICKJS)
    matter_register_component(matter_host_quickjs)
endif()
matter_write_component_graph("${CMAKE_CURRENT_BINARY_DIR}/component-graph.txt")
if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_subdirectory(tests/independent)
    add_executable(matter_contracts_legacy_checks tests/legacy_compatibility.cpp)
    target_link_libraries(matter_contracts_legacy_checks PRIVATE matter_contracts)
    add_test(NAME matter_contracts_legacy_checks COMMAND matter_contracts_legacy_checks)
    # Every C++ header must be self-contained, including the C ABI in C++ mode.
    foreach(header contracts jobs generation hosts graphics terrain abi)
        set(header_check "${CMAKE_CURRENT_BINARY_DIR}/header-${header}.cpp")
        file(WRITE "${header_check}" "#include <matter/contracts/${header}.h>\n")
        add_library(matter_header_${header} OBJECT "${header_check}")
        target_link_libraries(matter_header_${header} PRIVATE matter_contracts)
    endforeach()
    add_executable(matter_contracts_consumer_tests tests/service_producers.cpp tests/service_consumers.cpp)
    target_link_libraries(matter_contracts_consumer_tests PRIVATE matter_contracts)
    add_test(NAME matter_contracts_consumer_tests COMMAND matter_contracts_consumer_tests)
    add_executable(matter_contracts_c_ownership_tests tests/abi_producer.c tests/abi_consumer.c)
    target_include_directories(matter_contracts_c_ownership_tests PRIVATE include)
    set_target_properties(matter_contracts_c_ownership_tests PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
    add_test(NAME matter_contracts_c_ownership_tests COMMAND matter_contracts_c_ownership_tests)
    add_executable(matter_contracts_tests tests/contracts_tests.cpp)
    target_link_libraries(matter_contracts_tests PRIVATE matter_contracts)
    add_test(NAME matter_contracts_tests COMMAND matter_contracts_tests)
    add_executable(matter_contracts_vectors tests/vectors.cpp)
    target_link_libraries(matter_contracts_vectors PRIVATE matter_contracts)
    find_program(MATTER_CONTRACTS_NODE NAMES node node.exe REQUIRED)
    add_test(NAME matter_contracts_js_vectors COMMAND ${MATTER_CONTRACTS_NODE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/cross_language.mjs $<TARGET_FILE:matter_contracts_vectors>)
    add_test(NAME matter_contracts_schema COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/schema_tests.py --vectors $<TARGET_FILE:matter_contracts_vectors>)
endif()
