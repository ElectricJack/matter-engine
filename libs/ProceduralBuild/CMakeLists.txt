cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(matter_build LANGUAGES C CXX)
    include(CTest)
endif()
if(NOT TARGET matter_contracts)
    set(matter_build_testing ${BUILD_TESTING})
    set(BUILD_TESTING OFF)
    add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../MatterContracts" "${CMAKE_CURRENT_BINARY_DIR}/contracts")
    set(BUILD_TESTING ${matter_build_testing})
endif()
add_library(matter_build STATIC src/coordinator.cpp)
find_package(Threads REQUIRED)
target_compile_features(matter_build PUBLIC cxx_std_17)
target_include_directories(matter_build PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(matter_build PUBLIC matter_contracts Threads::Threads)
if(MSVC)
    target_compile_options(matter_build PRIVATE /W4 /WX /permissive- /utf-8)
else()
    target_compile_options(matter_build PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
    add_executable(matter_build_tests tests/coordinator_tests.cpp)
    target_link_libraries(matter_build_tests PRIVATE matter_build Threads::Threads)
    foreach(fixture graph source_graph cache_keys cache controls budgets)
        add_test(NAME matter_build_${fixture} COMMAND matter_build_tests ${fixture})
        set_tests_properties(matter_build_${fixture} PROPERTIES TIMEOUT 30)
    endforeach()
endif()
