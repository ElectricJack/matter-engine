cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(MatterAssetPhysics VERSION 1.0.0 LANGUAGES C CXX)
    include(CTest)
endif()
get_filename_component(_physics_repo "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(NOT TARGET matter_contracts)
    set(_physics_testing ${BUILD_TESTING})
    set(BUILD_TESTING OFF)
    add_subdirectory("${_physics_repo}/libs/MatterContracts" "${CMAKE_CURRENT_BINARY_DIR}/contracts")
    set(BUILD_TESTING ${_physics_testing})
endif()
option(MATTER_PHYSICS_BOX3D "Enable the optional native Box3d settling backend" OFF)
option(MATTER_PHYSICS_TERRAIN_ADAPTER "Build the native terrain candidate adapter" ON)
add_library(matter_collision STATIC src/collision.cpp)
target_link_libraries(matter_collision PUBLIC matter_contracts)
target_include_directories(matter_collision PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
add_library(matter_physics_generation STATIC src/settling.cpp)
target_link_libraries(matter_physics_generation PUBLIC matter_collision)
target_compile_definitions(matter_physics_generation PRIVATE MATTER_PHYSICS_BOX3D=$<BOOL:${MATTER_PHYSICS_BOX3D}>)
foreach(target matter_collision matter_physics_generation)
    target_compile_features(${target} PUBLIC cxx_std_17)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
endforeach()
if(MATTER_PHYSICS_BOX3D)
    if(EMSCRIPTEN)
        message(FATAL_ERROR "Box3d settling is a native-only capability; load baked products in WASM")
    endif()
    if(TARGET matter_box3d)
        set(_physics_solver matter_box3d)
    elseif(TARGET box3d)
        set(_physics_solver box3d)
    else()
        # Vendor's local source list, no samples/downloads and no copied sources.
        # Set its project metadata locally; avoid its top-level CRT override.
        set(PROJECT_VERSION 0.1.0)
        set(PROJECT_VERSION_MAJOR 0)
        add_subdirectory("${_physics_repo}/third_party/box3d/src" "${CMAKE_CURRENT_BINARY_DIR}/box3d")
        set(_physics_solver box3d)
    endif()
    # The settle core lives in this module; bake_trace.cpp is the engine's bake
    # observer, compiled in place. No monolithic engine archive, ECS, JS, or
    # graphics dependency.
    add_library(matter_settle_native STATIC
        src/tileset_settle.cpp
        "${_physics_repo}/MatterEngine3/src/bake_trace.cpp")
    target_compile_features(matter_settle_native PUBLIC cxx_std_17)
    target_include_directories(matter_settle_native PUBLIC "${_physics_repo}/MatterEngine3/src")
    target_link_libraries(matter_settle_native PUBLIC ${_physics_solver})
    target_link_libraries(matter_physics_generation PRIVATE matter_settle_native)
    # Content identity includes source + headers and the shared settle core.
    file(GLOB_RECURSE _solver_inputs CONFIGURE_DEPENDS
        "${_physics_repo}/third_party/box3d/src/*.c" "${_physics_repo}/third_party/box3d/src/*.h"
        "${_physics_repo}/third_party/box3d/src/*.inl" "${_physics_repo}/third_party/box3d/include/*.h")
    list(APPEND _solver_inputs "${CMAKE_CURRENT_SOURCE_DIR}/src/tileset_settle.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/tileset_settle.h")
    list(SORT _solver_inputs)
    set(_solver_hashes "")
    foreach(input IN LISTS _solver_inputs)
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${input}")
        file(SHA256 "${input}" digest)
        string(APPEND _solver_hashes "${digest}\n")
    endforeach()
    string(SHA256 MATTER_SETTLE_SOURCE_SHA256 "${_solver_hashes}")
    set(MATTER_SETTLE_BUILD_PROFILE "${CMAKE_SYSTEM_NAME}/${CMAKE_SYSTEM_PROCESSOR}/${CMAKE_C_COMPILER_ID}-${CMAKE_C_COMPILER_VERSION}/${CMAKE_BUILD_TYPE};double=${BOX3D_DOUBLE_PRECISION};simd-off=${BOX3D_DISABLE_SIMD}")
    configure_file(src/solver_identity.h.in solver_identity.h @ONLY)
    target_include_directories(matter_physics_generation PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
endif()
if(MATTER_PHYSICS_TERRAIN_ADAPTER)
    add_subdirectory("${_physics_repo}/MatterEngine3/src/component_adapters/collision" "${CMAKE_CURRENT_BINARY_DIR}/terrain-adapter")
endif()
if(BUILD_TESTING)
    set(_physics_testing ${BUILD_TESTING})
    set(BUILD_TESTING OFF)
    if(NOT TARGET matter_asset_archive)
        set(MATTER_ASSET_GLTF OFF)
        set(MATTER_ASSET_STORAGE OFF)
        set(MATTER_ASSET_LEGACY OFF)
        add_subdirectory("${_physics_repo}/libs/AssetPipeline" "${CMAKE_CURRENT_BINARY_DIR}/archive")
    endif()
    set(BUILD_TESTING ${_physics_testing})
    add_executable(matter_collision_tests tests/collision_tests.cpp)
    target_link_libraries(matter_collision_tests PRIVATE matter_physics_generation matter_asset_archive)
    target_compile_definitions(matter_collision_tests PRIVATE MATTER_PHYSICS_BOX3D=$<BOOL:${MATTER_PHYSICS_BOX3D}>)
    add_test(NAME physics.products COMMAND matter_collision_tests products)
    add_test(NAME physics.generation COMMAND matter_collision_tests generation)
    if(MATTER_PHYSICS_BOX3D)
        target_link_libraries(matter_collision_tests PRIVATE matter_settle_native)
        add_test(NAME physics.write_baked COMMAND matter_collision_tests write "${CMAKE_CURRENT_BINARY_DIR}/settled.asset")
    endif()
    if(MATTER_PHYSICS_TERRAIN_ADAPTER)
        add_executable(matter_collision_terrain_tests tests/terrain_tests.cpp)
        target_link_libraries(matter_collision_terrain_tests PRIVATE matter_collision_terrain_adapter)
        target_include_directories(matter_collision_terrain_tests PRIVATE "${_physics_repo}/MatterEngine3/src" "${_physics_repo}/MatterEngine3/include")
        add_test(NAME physics.terrain COMMAND matter_collision_terrain_tests)
    endif()
endif()
