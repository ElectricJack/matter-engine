cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(MatterAnimationAsset VERSION 1.0.0 LANGUAGES C CXX)
    include(CTest)
endif()
get_filename_component(MATTER_ANIMATION_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(NOT TARGET matter_contracts)
    set(_animation_testing ${BUILD_TESTING})
    set(BUILD_TESTING OFF)
    add_subdirectory("${MATTER_ANIMATION_ROOT}/libs/MatterContracts" "${CMAKE_CURRENT_BINARY_DIR}/contracts")
    set(BUILD_TESTING ${_animation_testing})
endif()
add_library(matter_animation_asset STATIC src/animation.cpp)
target_include_directories(matter_animation_asset PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(matter_animation_asset PUBLIC matter_contracts)
target_compile_features(matter_animation_asset PUBLIC cxx_std_17)
if(MSVC)
    target_compile_options(matter_animation_asset PRIVATE /W4 /WX /permissive- /utf-8)
else()
    target_compile_options(matter_animation_asset PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(MATTER_ANIMATION_NATIVE "Build explicit native authored/Ozz compatibility adapter" OFF)
if(BUILD_TESTING)
    add_executable(matter_animation_asset_tests tests/animation_tests.cpp)
    target_link_libraries(matter_animation_asset_tests PRIVATE matter_animation_asset)
    add_test(NAME matter_animation_asset COMMAND matter_animation_asset_tests)
endif()
if(MATTER_ANIMATION_NATIVE AND NOT TARGET matter_animation_native)
    add_subdirectory("${MATTER_ANIMATION_ROOT}/MatterEngine3/src/component_adapters/animation"
                     "${CMAKE_CURRENT_BINARY_DIR}/native")
endif()
