cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(matter_terrain_coordinator LANGUAGES C CXX)
    include(CTest)
endif()
get_filename_component(MATTER_COORDINATOR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
if(NOT TARGET matter_contracts)
    set(coordinator_testing ${BUILD_TESTING})
    set(BUILD_TESTING OFF)
    add_subdirectory("${MATTER_COORDINATOR_ROOT}/libs/MatterContracts" "${CMAKE_CURRENT_BINARY_DIR}/contracts")
    set(BUILD_TESTING ${coordinator_testing})
endif()
add_library(matter_terrain_coordinator STATIC src/coordinator.cpp src/cache.cpp)
target_compile_features(matter_terrain_coordinator PUBLIC cxx_std_17)
target_include_directories(matter_terrain_coordinator PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(matter_terrain_coordinator PUBLIC matter_contracts)
if(MSVC)
    target_compile_options(matter_terrain_coordinator PRIVATE /W4 /WX /permissive- /utf-8)
else()
    target_compile_options(matter_terrain_coordinator PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    add_executable(matter_terrain_coordinator_tests tests/coordinator_tests.cpp)
    target_link_libraries(matter_terrain_coordinator_tests PRIVATE matter_terrain_coordinator Threads::Threads)
    foreach(fixture epochs lifetimes cache pressure failures graphics thread)
        add_test(NAME terrain_coordinator_${fixture} COMMAND matter_terrain_coordinator_tests ${fixture})
        set_tests_properties(terrain_coordinator_${fixture} PROPERTIES TIMEOUT 30)
    endforeach()
endif()
