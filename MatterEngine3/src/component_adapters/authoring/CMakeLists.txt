cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(matter_authoring_adapter LANGUAGES C CXX)
    include(CTest)
endif()
get_filename_component(MATTER_AUTHORING_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
# Each component keeps its own guarded contracts load; their tests stay theirs.
set(authoring_testing ${BUILD_TESTING})
set(BUILD_TESTING OFF)
set(MATTER_GEOMETRY_TESTS OFF)
if(NOT TARGET matter_contracts)
    add_subdirectory("${MATTER_AUTHORING_ROOT}/libs/MatterContracts" "${CMAKE_CURRENT_BINARY_DIR}/contracts")
endif()
if(NOT TARGET matter_geometry)
    add_subdirectory("${MATTER_AUTHORING_ROOT}/libs/ProceduralGeometry" "${CMAKE_CURRENT_BINARY_DIR}/geometry")
endif()
if(NOT TARGET matter_build)
    add_subdirectory("${MATTER_AUTHORING_ROOT}/libs/ProceduralBuild" "${CMAKE_CURRENT_BINARY_DIR}/build")
endif()
if(NOT TARGET matter_host_quickjs)
    add_subdirectory("${MATTER_AUTHORING_ROOT}/MatterEngine3/src/script/portable_host" "${CMAKE_CURRENT_BINARY_DIR}/host")
endif()
# Graphics is the e2e fixture's consumer only; the adapter library never links it.
if(authoring_testing AND NOT TARGET matter_graphics)
    add_subdirectory("${MATTER_AUTHORING_ROOT}/libs/MatterGraphics" "${CMAKE_CURRENT_BINARY_DIR}/graphics")
endif()
set(BUILD_TESTING ${authoring_testing})
add_library(matter_authoring_adapter STATIC src/pipeline.cpp)
target_compile_features(matter_authoring_adapter PUBLIC cxx_std_17)
target_include_directories(matter_authoring_adapter PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(matter_authoring_adapter PUBLIC matter_build matter_geometry matter_host_quickjs)
if(MSVC)
    target_compile_options(matter_authoring_adapter PRIVATE /W4 /WX /permissive- /utf-8)
else()
    target_compile_options(matter_authoring_adapter PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
    add_executable(matter_authoring_e2e_tests tests/authoring_e2e_tests.cpp)
    target_link_libraries(matter_authoring_e2e_tests PRIVATE matter_authoring_adapter)
    target_compile_definitions(matter_authoring_e2e_tests PRIVATE MATTER_AUTHORING_REPO="${MATTER_AUTHORING_ROOT}")
    get_target_property(authoring_flags matter_authoring_adapter COMPILE_OPTIONS)
    target_compile_options(matter_authoring_e2e_tests PRIVATE ${authoring_flags})
    set(authoring_fixtures authored geometry material mirror shared seeded errors dedup)
    if(TARGET matter_graphics_vulkan)
        target_link_libraries(matter_authoring_e2e_tests PRIVATE matter_graphics_vulkan)
        target_include_directories(matter_authoring_e2e_tests PRIVATE "${MATTER_AUTHORING_ROOT}/libs/MatterGraphics/examples")
        target_compile_definitions(matter_authoring_e2e_tests PRIVATE MATTER_AUTHORING_VULKAN=1)
        list(APPEND authoring_fixtures graphics)
    endif()
    foreach(fixture ${authoring_fixtures})
        add_test(NAME authoring_e2e_${fixture} COMMAND matter_authoring_e2e_tests ${fixture})
        set_tests_properties(authoring_e2e_${fixture} PROPERTIES TIMEOUT 180)
    endforeach()
endif()
