# Embed source-of-truth modules, not maintained copies or runtime file reads.
# Byte arrays also avoid MSVC's string-literal size limit.
set(bundle "#pragma once\n#include <cstddef>\nnamespace matter::portable_host::embedded {\n")
set(entries "")
function(matter_host_embed id path symbol)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${path}")
  file(READ "${path}" bytes HEX)
  string(REGEX REPLACE "(..)" "0x\\1," bytes "${bytes}")
  set(bundle "${bundle}inline const unsigned char ${symbol}[] = {${bytes}0};\n" PARENT_SCOPE)
  set(entries "${entries}{\"${id}\", ${symbol}, sizeof(${symbol}) - 1},\n" PARENT_SCOPE)
endfunction()
foreach(module index values parameters rng recipe legacy)
  matter_host_embed("matter-internal/${module}.mjs"
    "${MATTER_HOST_ROOT}/packages/matter-authoring/${module}.mjs" "sdk_${module}")
endforeach()
matter_host_embed("matter-internal/contracts.mjs"
  "${MATTER_HOST_ROOT}/libs/MatterContracts/examples/contracts.mjs" contracts)
matter_host_embed("matter-internal/bridge.mjs" "${CMAKE_CURRENT_SOURCE_DIR}/bridge.mjs" bridge)
matter_host_embed("matter-internal/setup.js" "${CMAKE_CURRENT_SOURCE_DIR}/setup.js" setup)
string(APPEND bundle "struct Module { const char *id; const unsigned char *source; size_t size; };\ninline const Module modules[] = {${entries}};\n}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/embedded_modules.h" "${bundle}")
