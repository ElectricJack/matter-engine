cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(MatterGeometryC VERSION 1.0.0 LANGUAGES C CXX)
    include(CTest)
endif()
option(MATTER_GEOMETRY_C_WASM_SMOKE "Also compile the frozen contract WASM fixtures" OFF)
option(MATTER_GEOMETRY_C_TESTS "Build the real C ABI consumer fixtures" ON)
if(EMSCRIPTEN)
    include(wasm/Profile.cmake)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../libs/MatterContracts/cmake/ComponentScaffolds.cmake")
# Contract/native geometry suites have independent builds. Do not register
# native launchers or the legacy surface baseline in a WASM consumer graph.
set(_binding_testing "${BUILD_TESTING}")
set(BUILD_TESTING OFF)
if(NOT TARGET matter_geometry)
    set(MATTER_GEOMETRY_TESTS OFF)
    matter_register_component(matter_geometry)
endif()
set(BUILD_TESTING "${_binding_testing}")
add_library(matter_geometry_c STATIC src/geometry_c.cpp src/wire.cpp)
target_compile_features(matter_geometry_c PUBLIC cxx_std_17)
target_include_directories(matter_geometry_c PUBLIC include)
target_link_libraries(matter_geometry_c PUBLIC matter_contracts matter_geometry)
if(MSVC)
    target_compile_options(matter_geometry_c PRIVATE /W4 /WX /permissive- /utf-8)
else()
    target_compile_options(matter_geometry_c PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(EMSCRIPTEN)
    include(wasm/ScalarHeaders.cmake)
    foreach(target matter_contracts matter_geometry matter_geometry_c)
        target_link_libraries(${target} PRIVATE matter_wasm_profile)
    endforeach()
    include(wasm/Module.cmake)
endif()
if(MATTER_GEOMETRY_C_TESTS)
    enable_testing()
    include(tests/Fixtures.cmake)
    add_executable(matter_geometry_c_tests tests/consumer.c)
    target_include_directories(matter_geometry_c_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    target_link_libraries(matter_geometry_c_tests PRIVATE matter_geometry_c)
    set_target_properties(matter_geometry_c_tests PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
    if(EMSCRIPTEN)
        set_target_properties(matter_geometry_c_tests PROPERTIES SUFFIX ".mjs")
        target_link_libraries(matter_geometry_c_tests PRIVATE matter_wasm_profile)
        target_link_options(matter_geometry_c_tests PRIVATE "-sEXPORTED_RUNTIME_METHODS=callMain,HEAPU8" "-sEXIT_RUNTIME=1" "-sSINGLE_FILE=1")
        add_test(NAME matter_geometry_c_consumer COMMAND "${MATTER_WASM_NODE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/wasm/run-smoke.mjs" "$<TARGET_FILE:matter_geometry_c_tests>")
    else()
        add_test(NAME matter_geometry_c_consumer COMMAND matter_geometry_c_tests)
        add_executable(matter_geometry_c_allocation_tests tests/allocation.cpp)
        target_include_directories(matter_geometry_c_allocation_tests PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
        target_link_libraries(matter_geometry_c_allocation_tests PRIVATE matter_geometry_c)
        add_test(NAME matter_geometry_c_allocation COMMAND matter_geometry_c_allocation_tests)
        set_tests_properties(matter_geometry_c_allocation PROPERTIES TIMEOUT 180)
    endif()
endif()
if(MATTER_GEOMETRY_C_WASM_SMOKE)
    if(NOT EMSCRIPTEN)
        message(FATAL_ERROR "WASM smoke requires Emscripten")
    endif()
    include(wasm/Smoke.cmake)
endif()
