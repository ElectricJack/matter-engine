cmake_minimum_required(VERSION 3.25)
project(MatterDemoRecipes LANGUAGES C CXX)
include(CTest)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(MATTER_GEOMETRY_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../../libs/ProceduralGeometry" geometry)
add_executable(matter_demo_native_recipes native_recipes.cpp)
target_link_libraries(matter_demo_native_recipes PRIVATE matter_geometry)
if(MSVC)
    target_compile_options(matter_demo_native_recipes PRIVATE /W4 /WX /permissive- /utf-8)
else()
    target_compile_options(matter_demo_native_recipes PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
    find_program(MATTER_RECIPE_NODE node REQUIRED)
    find_program(MATTER_RECIPE_TSC tsc REQUIRED)
    add_test(NAME matter_demo_recipes COMMAND "${MATTER_RECIPE_NODE}" "${CMAKE_CURRENT_LIST_DIR}/recipes.mts" "$<TARGET_FILE:matter_demo_native_recipes>")
    # Includes Node startup, SDK control regressions and three native catalogs.
    # The native executable enforces the separate 2-second in-memory budget.
    set_tests_properties(matter_demo_recipes PROPERTIES TIMEOUT 60)
    add_test(NAME matter_demo_recipe_types COMMAND "${MATTER_RECIPE_TSC}" -p "${CMAKE_CURRENT_LIST_DIR}/tsconfig.json")
endif()
